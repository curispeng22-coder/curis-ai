41:   function parse() {
42:     const raw = location.hash || '#/ceo';
43:     const [path, query] = raw.split('?');
44:     const q = {}; (query || '').split('&').filter(Boolean).forEach(kv => { const [k, v] = kv.split('='); q[k] = decodeURIComponent(v || ''); });
45:     for (const [re, fn] of ROUTES) { const m = path.match(re); if (m) { const r = fn(m); r.q = q; return r; } }
46:     return { page: 'ceo', q };
47:   }
